cmake_minimum_required(VERSION 3.22)
project(lwm2m_native_checks C)
set(CMAKE_C_STANDARD 11)
# Only the loopback wire-test peer uses NoSec; firmware requires X.509 DTLS.
set(WITH_UNSECURE_CONNECTIONS ON CACHE BOOL "" FORCE)
set(ANJAY_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../.deps/anjay" CACHE PATH "Pinned Anjay source tree" FORCE)
set(DTLS_BACKEND openssl CACHE STRING "" FORCE)
set(WITH_TEST OFF CACHE BOOL "" FORCE)
set(WITH_DEMO OFF CACHE BOOL "" FORCE)
set(WITH_LIBRARY_SHARED OFF CACHE BOOL "" FORCE)
set(WITH_DOWNLOADER OFF CACHE BOOL "" FORCE)
set(WITH_BOOTSTRAP OFF CACHE BOOL "" FORCE)
set(WITH_LWM2M12 OFF CACHE BOOL "" FORCE)
# Upstream connection-status code references disabled bootstrap state.
set(WITH_CONN_STATUS_API OFF CACHE BOOL "" FORCE)
set(WITH_COMMUNICATION_TIMESTAMP_API ON CACHE BOOL "" FORCE)
set(WITH_MODULE_fw_update ON CACHE BOOL "" FORCE)
set(WITH_MODULE_advanced_fw_update OFF CACHE BOOL "" FORCE)
set(WITH_MODULE_sw_mgmt OFF CACHE BOOL "" FORCE)
set(WITH_MODULE_ipso_objects OFF CACHE BOOL "" FORCE)
set(WITH_MODULE_ipso_objects_v2 OFF CACHE BOOL "" FORCE)
set(WITH_MODULE_factory_provisioning OFF CACHE BOOL "" FORCE)
set(WITH_ANJAY_TRACE_LOGS OFF CACHE BOOL "" FORCE)
set(WITH_AVS_COAP_TRACE_LOGS OFF CACHE BOOL "" FORCE)
set(WITH_INTERNAL_TRACE OFF CACHE BOOL "" FORCE)
add_subdirectory("${ANJAY_SOURCE}" anjay)
add_executable(test_lwm2m ../native/test_lwm2m.c)
target_include_directories(test_lwm2m BEFORE PRIVATE ../native ../../main)
target_compile_options(test_lwm2m PRIVATE -g -fsanitize=address,undefined)
target_link_options(test_lwm2m PRIVATE -fsanitize=address,undefined)
target_link_libraries(test_lwm2m PRIVATE anjay)
add_executable(observe_probe observe_probe.c ../../main/lwm2m/lwm2m_sensor.c)
target_include_directories(observe_probe PRIVATE ../native ../../main)
target_link_libraries(observe_probe PRIVATE anjay)
add_executable(firmware_probe firmware_probe.c ../../main/lwm2m/lwm2m_firmware.c)
target_include_directories(firmware_probe PRIVATE ../native ../../main)
target_link_libraries(firmware_probe PRIVATE anjay)
